function updateCharts(data) { const mesesNomes = ["Jan", "Fev", "Mar", "Abr", "Mai", "Jun", "Jul", "Ago", "Set", "Out", "Nov", "Dez"]; const mesesContagem = {}; data.forEach(d => { if (d.data) { const str = String(d.data).trim().replace(/\//g, '-'); const parts = str.split('-'); if (parts.length === 3) { let ano, mesIdx; // Se for AAAA-MM-DD if (parts[0].length === 4) { ano = parts[0]; mesIdx = parseInt(parts[1]) - 1; } else { // Se for DD-MM-AAAA ano = parts[2]; mesIdx = parseInt(parts[1]) - 1; } if (!isNaN(mesIdx) && mesIdx >= 0 && mesIdx < 12) { const chave = `${mesesNomes[mesIdx]}/${ano.slice(-2)}`; mesesContagem[chave] = (mesesContagem[chave] || 0) + 1; } } } }); mesesChart.data.labels = Object.keys(mesesContagem); mesesChart.data.datasets[0].data = Object.values(mesesContagem); mesesChart.update(); const realizados = data.filter(d => String(d.status).toLowerCase() === 'sim').length; const pendentes = data.length - realizados; statusChart.data.datasets[0].data = [realizados, pendentes]; statusChart.update(); const empresas = {}; data.forEach(d => { if(d.empresa) empresas[d.empresa] = (empresas[d.empresa] || 0) + 1; }); const sortedEmpresas = Object.entries(empresas).sort((a, b) => b[1] - a[1]).slice(0, 5); empresaChart.data.labels = sortedEmpresas.map(item => item[0]); empresaChart.data.datasets[0].data = sortedEmpresas.map(item => item[1]); empresaChart.update(); const estados = {}; data.forEach(d => { if(d.estado) estados[d.estado] = (estados[d.estado] || 0) + 1; }); const sortedEstados = Object.entries(estados).sort((a, b) => b[1] - a[1]).slice(0, 5); estadoChart.data.labels = sortedEstados.map(item => item[0]); estadoChart.data.datasets[0].data = sortedEstados.map(item => item[1]); estadoChart.update(); }
